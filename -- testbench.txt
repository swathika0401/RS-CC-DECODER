    signal rs_encoded : std_logic_vector(2039 downto 0);

    signal rs_start : std_logic := '0';
    signal rs_codeword : std_logic_vector(2039 downto 0) :=

    signal rs_done : std_logic;

        rs_start <= '0';
        rs_reset <= '1';

        wait until falling_edge(rs_clk);

        rs_reset <= '0';
        wait until falling_edge(rs_clk);

        --------------------------------------------------
        -- Encode non-zero message
        --------------------------------------------------
        rs_enc_start <= '1';

        wait until falling_edge(rs_clk);
        rs_enc_start <= '0';

        timeout_count := 0;

        while rs_enc_done /= '1' and timeout_count < 10 loop
            wait until rising_edge(rs_clk);
            wait for 0 ns;
            timeout_count := timeout_count + 1;
        end loop;

        assert rs_enc_done = '1'
            report "ERROR: RS encoder did not complete"
            severity failure;

        wait for 1 ns;
        original_cw := rs_encoded;

        assert original_cw /= (original_cw'range => '0')
            report "FAIL: encoder produced an all-zero codeword"
            severity error;

        report "Reference encoder completed for non-zero data"
            severity note;

        report "Encoder output captured" severity note;

        report "First encoded symbol = " &
            integer'image(
                to_integer(unsigned(original_cw(2039 downto 2032)))
            )
            severity note;

        --------------------------------------------------
        -- RS TEST 0: Clean non-zero codeword
        --------------------------------------------------
        rs_start <= '0';
        rs_reset <= '1';
        rs_codeword <= original_cw;

        wait until falling_edge(rs_clk);
        wait until falling_edge(rs_clk);

        rs_reset <= '0';
        wait until falling_edge(rs_clk);

        rs_start <= '1';
        wait until falling_edge(rs_clk);
        rs_start <= '0';

        wait until rs_done = '1';
        wait for 1 ns;

        if rs_error_detect = '0' and
           rs_correction_valid = '1' and
           rs_corrected_codeword = original_cw then

            report "PASS: clean non-zero RS codeword"
                severity note;
        else
            report "FAIL: clean non-zero RS codeword; error_detect=" &
                   std_logic'image(rs_error_detect) &
                   ", correction_valid=" &
                   std_logic'image(rs_correction_valid)
                severity error;
        end if;

        --------------------------------------------------
        -- RS TESTS: 1, 2, 4 and 8 symbol errors
        --------------------------------------------------
        for test_no in 1 to 4 loop

            case test_no is
                when 1 => error_count := 1;
                when 2 => error_count := 2;
                when 3 => error_count := 4;
                when others => error_count := 8;

            rs_start <= '0';
            rs_reset <= '1';
            rs_codeword <= original_cw;

            wait until falling_edge(rs_clk);
            wait until falling_edge(rs_clk);

            rs_reset <= '0';
            wait until falling_edge(rs_clk);

            -- Inject symbol errors

            for i in 0 to error_count - 1 loop
                cw(2039 - 8*i downto 2032 - 8*i) :=
                    cw(2039 - 8*i downto 2032 - 8*i) xor
                    std_logic_vector(to_unsigned(i + 1, 8));
            end loop;

            rs_codeword <= cw;

            wait until falling_edge(rs_clk);
            rs_start <= '1';

            wait until falling_edge(rs_clk);
            rs_start <= '0';

            wait until rs_done = '1';
            wait for 1 ns;

            assert rs_error_detect = '1'
                report "FAIL: error not detected for " &
                       integer'image(error_count) & " symbol errors"
                severity error;

            assert rs_correction_valid = '1'
                report "FAIL: correction_valid for " &
                       integer'image(error_count) & " symbol errors"
                severity error;

            assert rs_corrected_codeword = original_cw
                report "FAIL: output mismatch for " &
                       integer'image(error_count) & " symbol errors"
                severity error;

            if rs_correction_valid = '1' and
               rs_corrected_codeword = original_cw then

                report "PASS: corrected " &
                       integer'image(error_count) &
                       " symbol errors in non-zero codeword"
                    severity note;
            end if;

        end loop;

        report "All CC and non-zero RS tests completed"
            severity note;

        wait;
    end process;

end architecture sim;

library ieee;
use ieee.std_logic_1164.all;
use std.env.all;

entity tb_cc_viterbi is
end entity tb_cc_viterbi;

architecture sim of tb_cc_viterbi is

    constant DATA_BITS : positive := 14;
    constant STEPS     : positive := DATA_BITS + 6;

    signal clk       : std_logic := '0';
    signal reset     : std_logic := '1';
    signal data_in   : std_logic := '0';
    signal valid_in  : std_logic := '0';

    signal code_out  : std_logic_vector(1 downto 0);
    signal valid_out : std_logic;

    signal rx_bits : std_logic_vector(2*STEPS-1 downto 0)
                     := (others => '0');

    signal decoded : std_logic_vector(STEPS-1 downto 0)
                     := (others => '0');

    signal dec_start : std_logic := '0';
    signal dec_busy  : std_logic;
    signal dec_done  : std_logic;

    constant MESSAGE : std_logic_vector(DATA_BITS-1 downto 0)
                       := "10110010000000";

    function bits_to_string(
        v : std_logic_vector
    ) return string is
        variable s : string(1 to v'length);
        variable k : integer := 1;
    begin
        for i in v'range loop
            if v(i) = '0' then
                s(k) := '0';
            elsif v(i) = '1' then
                s(k) := '1';
            else
                s(k) := 'X';
            end if;
            k := k + 1;
        end loop;
        return s;
    end function;

begin

    clk <= not clk after 5 ns;

    enc: entity work.cc_encoder_ref
        port map (
            clk       => clk,
            reset     => reset,
            data_in   => data_in,
            valid_in  => valid_in,
            code_out  => code_out,
            valid_out => valid_out
        );

    dec: entity work.viterbi_frame_decoder
        generic map (
            FRAME_BITS => STEPS
        )
        port map (
            clk          => clk,
            reset        => reset,
            start        => dec_start,
            rx_bits      => rx_bits,
            decoded_bits => decoded,
            busy         => dec_busy,
            done         => dec_done
        );

    stimulus: process
        variable frame   : std_logic_vector(STEPS-1 downto 0);
        variable encoded : std_logic_vector(2*STEPS-1 downto 0);
    begin

        report "Starting CC + Viterbi single-bit error test"
            severity note;

        frame := MESSAGE & "000000";

        wait for 20 ns;
        wait until falling_edge(clk);
        reset <= '0';

        -- Encode message and six zero tail bits
        for i in 0 to STEPS-1 loop
            data_in  <= frame(STEPS-1-i);
            valid_in <= '1';

            wait until rising_edge(clk);
            wait for 1 ns;

            assert valid_out = '1'
                report "FAIL: encoder valid_out was not asserted"
                severity failure;

            encoded(2*STEPS-1-2*i downto
                    2*STEPS-2-2*i) := code_out;

            wait until falling_edge(clk);
        end loop;

        valid_in <= '0';

        -- Inject one channel-bit error
        encoded(10) := not encoded(10);

        report "Single-bit channel error injected"
            severity note;

        -- Supply the corrupted encoded frame
        rx_bits <= encoded;

        wait until falling_edge(clk);
        dec_start <= '1';

        wait until falling_edge(clk);
        dec_start <= '0';

        -- Wait until Viterbi decoding is complete
        wait until dec_done = '1';
        wait for 1 ns;

        report "Expected: " & bits_to_string(MESSAGE)
            severity note;

        report "Decoded : " &
            bits_to_string(decoded(STEPS-1 downto 6))
            severity note;

        assert decoded(STEPS-1 downto 6) = MESSAGE
            report "FAIL: decoded message differs from original"
            severity failure;

        assert decoded(5 downto 0) = "000000"
            report "FAIL: tail bits were not recovered"
            severity failure;

        report "PASS: CC encoder + Viterbi decoder recovered message after one-bit error"
            severity note;

        stop;
        wait;
    end process;

end architecture sim;

    signal ref_rs_done : std_logic;

    signal ref_data  : std_logic := '0';
    signal ref_valid : std_logic := '0';
    signal ref_code  : std_logic_vector(1 downto 0);
    signal ref_valid_out : std_logic;

    signal ref_bits : std_logic_vector(4079 downto 0)
                      := (others => '0');

    signal ref_index : integer range 0 to 2040 := 0;
    signal ref_count : integer range 0 to 2040 := 0;
    signal ref_active : std_logic := '0';
    signal ref_complete : std_logic := '0';

begin

    clk <= not clk after 5 ns;

    -- Transmitter under test
    dut : entity work.tx_chain
        port map (
            clk     => clk,
            reset   => reset,
            start   => start,
            data_in => data_in,
            tx_bits => tx_bits,
            busy    => busy,
            done    => done
        );

    -- Reference RS encoder
    ref_rs : entity work.rs_reference_encoder
        port map (
            clk          => clk,
            reset        => reset,
            start        => start,
            data_in      => data_in,
            codeword_out => ref_rs_word,
            busy         => ref_rs_busy,
            done         => ref_rs_done
        );

    -- Reference interleaver
    ref_int : entity work.interleaver
        port map (
            data_in  => ref_rs_word,
            data_out => ref_mixed
        );

    -- Reference CC encoder
    ref_cc : entity work.cc_encoder_ref
        port map (
            clk       => clk,
            reset     => reset,
            data_in   => ref_data,
            valid_in  => ref_valid,
            code_out  => ref_code,
            valid_out => ref_valid_out
        );

    -- Reference serializer and output capture
    process(clk)
    begin
        if rising_edge(clk) then

            if reset = '1' then

                ref_valid    <= '0';
                ref_index    <= 0;

        assert tx_bits /= (tx_bits'range => '0')

            severity failure;

        assert tx_bits = ref_bits
            report "FAIL: transmitter output differs from reference"

        report "PASS: transmitter output matches reference"
            severity note;

        wait;

library ieee;
use ieee.std_logic_1164.all;
use ieee.numeric_std.all;

entity tb_viterbi_frame is
end entity tb_viterbi_frame;

architecture sim of tb_viterbi_frame is

    constant FRAME_BITS : positive := 2040;

    signal clk          : std_logic := '0';
    signal reset        : std_logic := '1';
    signal start        : std_logic := '0';

    signal data_in      : std_logic := '0';
    signal encoder_valid: std_logic := '0';
    signal code_out     : std_logic_vector(1 downto 0);
    signal code_valid   : std_logic;

    signal rx_bits      : std_logic_vector(2*FRAME_BITS-1 downto 0)
                          := (others => '0');

    signal decoded_bits : std_logic_vector(FRAME_BITS-1 downto 0);
    signal busy         : std_logic;
    signal done         : std_logic;

    signal frame        : std_logic_vector(FRAME_BITS-1 downto 0)
                          := (others => '0');

begin

    clk <= not clk after 5 ns;

    encoder_inst : entity work.cc_encoder_ref
        port map (
            clk       => clk,
            reset     => reset,
            data_in   => data_in,
            valid_in  => encoder_valid,
            code_out  => code_out,
            valid_out => code_valid
        );

    decoder_inst : entity work.viterbi_frame_decoder
        generic map (
            FRAME_BITS => FRAME_BITS
        )
        port map (
            clk          => clk,
            reset        => reset,
            start        => start,
            rx_bits      => rx_bits,
            decoded_bits => decoded_bits,
            busy         => busy,
            done         => done
        );

    stimulus : process
        variable encoded_count : integer := 0;
    begin

        report "TEST START" severity note;

        reset <= '1';
        encoder_valid <= '0';

        wait for 30 ns;
        wait until falling_edge(clk);
        reset <= '0';

        -- Generate deterministic frame
        for i in 0 to FRAME_BITS-1 loop
            if (i mod 7 = 0) or (i mod 11 = 0) then
                frame(FRAME_BITS-1-i) <= '1';
            else
                frame(FRAME_BITS-1-i) <= '0';
            end if;
        end loop;

        wait until falling_edge(clk);

        -- Enable encoder for exactly FRAME_BITS input bits
        encoded_count := 0;
        encoder_valid <= '1';

        for i in 0 to FRAME_BITS-1 loop

            data_in <= frame(FRAME_BITS-1-i);

            wait until rising_edge(clk);
            wait for 1 ns;

            if code_valid = '1' then
                rx_bits(
                    2*FRAME_BITS-1-2*encoded_count downto
                    2*FRAME_BITS-2-2*encoded_count
                ) <= code_out;

                encoded_count := encoded_count + 1;
            end if;

        end loop;

        encoder_valid <= '0';
        data_in <= '0';

        assert encoded_count = FRAME_BITS
            report "FAIL: incorrect encoded bit count"
            severity failure;

        report "Encoder completed; starting decoder"
            severity note;

        wait until falling_edge(clk);
        start <= '1';

        wait until falling_edge(clk);
        start <= '0';

        -- Prevent the test from hanging indefinitely
        wait until done = '1' for 100 us;

        if done /= '1' then
            assert false
                report "FAIL: Viterbi decoder timed out"
                severity failure;
        end if;

        wait for 1 ns;

        assert decoded_bits = frame
            report "FAIL: decoded frame differs from input"
            severity failure;

        report "PASS: original frame recovered successfully"
            severity note;

        wait;
    end process;

end architecture sim;

        port map (
            clk               => clk,
            reset             => reset,
            start             => rx_start,
            rx_bits           => tx_bits_with_error,

        rx_start <= '1';

        wait until rising_edge(clk);
        rx_start <= '0';

        -- Wait for receiver completion
        wait until rx_done = '1' for 2 ms;

        if rx_done /= '1' then
            assert false
                report "FAIL: depth-8 receiver timed out"
                severity failure;
            std.env.stop;
            wait;
        end if;

        wait for 1 ns;

        -- Compare recovered data with original input
        assert rx_data = original_data
            report "FAIL: recovered data differs after single-bit error"
            severity failure;

        report "PASS: depth-8 receiver recovered original data after single-bit error"
            severity note;

        std.env.stop;
        wait;

    end process;

end architecture sim;
